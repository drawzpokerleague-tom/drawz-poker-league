rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    match /players/{id} {
      allow read: if true;
      allow write: if signedIn();
    }

    match /tournaments/{id} {
      allow read: if true;
      allow write: if signedIn();
    }

    match /announcements/{id} {
      allow read: if true;
      allow write: if signedIn();
    }

    match /results/{id} {
      allow read: if true;
      allow write: if signedIn();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
